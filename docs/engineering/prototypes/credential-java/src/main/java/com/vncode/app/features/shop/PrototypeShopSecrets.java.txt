package com.vncode.app.features.shop;

import com.vncode.app.shared.credential.PrototypeCredentialAccess;
import com.vncode.app.shared.credential.PrototypeCredentialReader;
import com.vncode.app.shared.credential.PrototypeCredentialWriter;
import java.io.IOException;
import java.sql.Connection;
import java.sql.SQLException;

/** Prototype-only boundary; callers never infer the on-disk format from a prefix. */
public final class PrototypeShopSecrets {
    private static final String KIND = "SHOP_API_KEY";
    private PrototypeShopSecrets() { }
    public static String read(Connection connection, int shopId, String stored) throws SQLException {
        try { return PrototypeCredentialReader.read(connection, PrototypeCredentialAccess.codec(), KIND, shopId, stored); }
        catch (IOException unavailable) { throw new CredentialUnavailableException(); }
    }
    /** Caller must commit the returned stored value and metadata in the same transaction. */
    public static String write(Connection connection, int shopId, String plain) throws SQLException {
        try { return PrototypeCredentialWriter.write(connection, PrototypeCredentialAccess.codec(), KIND, shopId, plain); }
        catch (IOException unavailable) { throw new CredentialUnavailableException(); }
    }
    public static final class CredentialUnavailableException extends IllegalStateException {
        public CredentialUnavailableException() { super("SHOP_CREDENTIALS_UNAVAILABLE"); }
    }
}
