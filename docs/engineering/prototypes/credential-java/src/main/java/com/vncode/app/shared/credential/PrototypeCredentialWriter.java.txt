package com.vncode.app.shared.credential;
import java.io.IOException;
import java.sql.*;
import java.util.Objects;

public final class PrototypeCredentialWriter {
    private PrototypeCredentialWriter() {}
    /** Caller must write returned data and commit metadata within the same transaction. */
    public static String write(Connection c,PrototypeCredentialCodec codec,String kind,long id,String raw)throws SQLException,IOException{
        PrototypeCredentialReader.requireKind(kind,id);
        if(c.getAutoCommit())throw new SQLException("CREDENTIAL_TRANSACTION_REQUIRED");
        String format=raw==null?"NULL":raw.isEmpty()?"EMPTY":"PROTECTED";
        String stored=format.equals("PROTECTED")?codec.protect(raw):raw;
        if(format.equals("PROTECTED")&&!Objects.equals(raw,codec.reveal(stored)))throw new IOException("CREDENTIAL_ROUNDTRIP_FAILED");
        try(var p=c.prepareStatement("INSERT INTO prototype_credential_formats(kind,owner_id,format) VALUES(?,?,?) ON CONFLICT(kind,owner_id) DO UPDATE SET format=excluded.format")){
            p.setString(1,kind);p.setLong(2,id);p.setString(3,format);p.executeUpdate();
        }
        return stored;
    }
}
