package com.vncode.app.shared.credential;

import com.vncode.app.shared.WindowsSecretProtector;
import java.io.IOException;

/** Prototype test injection; this global arrangement is not a production security boundary. */
public final class PrototypeCredentialAccess {
    private static final ThreadLocal<PrototypeCredentialCodec> TEST_CODEC = new ThreadLocal<>();
    private static final PrototypeCredentialCodec SYSTEM_CODEC = new PrototypeCredentialCodec() {
            final WindowsSecretProtector protector=new WindowsSecretProtector("VN code shop OMS prototype",65536);
            public void ensureAvailable()throws IOException{if(!WindowsSecretProtector.supported())throw new IOException("WINDOWS_DPAPI_REQUIRED");}
            public String protect(String plain)throws IOException{return protector.protect(plain);}
            public String reveal(String value)throws IOException{return protector.reveal(value);}
        };
    private PrototypeCredentialAccess() {}
    public static PrototypeCredentialCodec codec() {
        var override=TEST_CODEC.get();
        return override!=null?override:SYSTEM_CODEC;
    }
    public static AutoCloseable withCodecForTest(PrototypeCredentialCodec codec) {
        if(codec==null||TEST_CODEC.get()!=null)throw new IllegalStateException("PROTOTYPE_CODEC_SCOPE_INVALID");
        TEST_CODEC.set(codec);
        Thread owner=Thread.currentThread();
        return ()->{if(Thread.currentThread()!=owner||TEST_CODEC.get()!=codec)throw new IllegalStateException("PROTOTYPE_CODEC_SCOPE_INVALID");TEST_CODEC.remove();};
    }
}
