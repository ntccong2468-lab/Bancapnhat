package com.vncode.app.shared.credential;
import java.io.IOException;
import java.sql.*;
import java.util.Set;

public final class PrototypeCredentialReader {
    static final Set<String> KINDS=Set.of("SHOP_API_KEY","ZNACK_SETTINGS_OMS","PIPELINE_OMS");
    private PrototypeCredentialReader() {}
    static void requireKind(String kind,long id){if(!KINDS.contains(kind)||id<=0)throw new IllegalArgumentException("CREDENTIAL_ID_INVALID");}
    public static String read(Connection c,PrototypeCredentialCodec codec,String kind,long id,String stored)throws SQLException,IOException{
        requireKind(kind,id);
        try(var p=c.prepareStatement("SELECT format FROM prototype_credential_formats WHERE kind=? AND owner_id=?")){
            p.setString(1,kind);p.setLong(2,id);
            try(var r=p.executeQuery()){
                if(!r.next())throw new IOException("CREDENTIAL_FORMAT_UNAVAILABLE");
                return switch(r.getString(1)){
                    case "MISSING" -> {if(!kind.equals("PIPELINE_OMS")||stored!=null)throw new IOException("CREDENTIAL_FORMAT_INVALID");yield null;}
                    case "NULL" -> {if(stored!=null)throw new IOException("CREDENTIAL_FORMAT_INVALID");yield null;}
                    case "EMPTY" -> {if(!"".equals(stored))throw new IOException("CREDENTIAL_FORMAT_INVALID");yield "";}
                    case "PROTECTED" -> {if(stored==null||stored.isEmpty())throw new IOException("CREDENTIAL_FORMAT_INVALID");yield codec.reveal(stored);}
                    default -> throw new IOException("CREDENTIAL_FORMAT_INVALID");
                };
            }
        }
    }
}
