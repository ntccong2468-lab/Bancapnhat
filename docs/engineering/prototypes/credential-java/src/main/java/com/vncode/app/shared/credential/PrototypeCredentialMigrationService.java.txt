package com.vncode.app.shared.credential;

import com.google.gson.JsonObject;
import com.google.gson.JsonParser;
import java.io.IOException;
import java.sql.*;
import java.util.*;

/** Experimental synthetic-data model. Caller must lock app data and verify backup before invocation. */
public final class PrototypeCredentialMigrationService {
    private PrototypeCredentialMigrationService() {}
    private record Row(String kind,long id,long shop,String source,String value,boolean missing) {}
    private record Converted(Row row,String format,String stored) {}
    public static void apply(Connection c,PrototypeCredentialCodec codec)throws SQLException,IOException {
        if(!c.getAutoCommit())throw new SQLException("CREDENTIAL_AUTOCOMMIT_REQUIRED");
        codec.ensureAvailable();
        List<Row> before=readRows(c);
        if(tableExists(c)){
            verifyComplete(c,codec,before);
            return;
        }
        List<Converted> converted=new ArrayList<>();
        for(Row row:before){
            String value=row.value();
            String format=row.missing()?"MISSING":value==null?"NULL":value.isEmpty()?"EMPTY":"PROTECTED";
            String stored=format.equals("PROTECTED")?codec.protect(value):value;
            if(format.equals("PROTECTED")&&!Objects.equals(value,codec.reveal(stored)))throw new IOException("CREDENTIAL_ROUNDTRIP_FAILED");
            converted.add(new Converted(row,format,stored));
        }
        // Do not invoke DPAPI inside the write lock. Re-check every source under BEGIN IMMEDIATE.
        try(Statement transaction=c.createStatement()){
            transaction.execute("BEGIN IMMEDIATE");
            try {
                if(tableExists(c)||!before.equals(readRows(c)))throw new SQLException("CREDENTIAL_SOURCE_CHANGED");
                transaction.execute("CREATE TABLE prototype_credential_formats(kind TEXT NOT NULL,owner_id INTEGER NOT NULL,format TEXT NOT NULL CHECK(format IN ('PROTECTED','EMPTY','NULL','MISSING')),PRIMARY KEY(kind,owner_id))");
                for(Converted item:converted){
                    Row row=item.row();
                    String sql=switch(row.kind()){
                        case "SHOP_API_KEY" -> "UPDATE shops SET api_key=? WHERE id=?";
                        case "ZNACK_SETTINGS_OMS" -> "UPDATE znack_settings SET oms_connection=? WHERE shop_id=?";
                        case "PIPELINE_OMS" -> "UPDATE znack_pipeline_connections SET settings_json=? WHERE pipeline_id=?";
                        default -> throw new IllegalStateException("CREDENTIAL_ID_INVALID");
                    };
                    String stored=row.kind().equals("PIPELINE_OMS")?(row.missing()?row.source():replaceOms(row.source(),item.stored())):item.stored();
                    try(var p=c.prepareStatement(sql)){p.setString(1,stored);p.setLong(2,row.id());if(p.executeUpdate()!=1)throw new SQLException("CREDENTIAL_SOURCE_CHANGED");}
                    try(var p=c.prepareStatement("INSERT INTO prototype_credential_formats(kind,owner_id,format) VALUES(?,?,?)")){
                        p.setString(1,row.kind());p.setLong(2,row.id());p.setString(3,item.format());p.executeUpdate();
                    }
                }
                transaction.execute("COMMIT");
            } catch(SQLException|IOException|RuntimeException failure) {
                try{transaction.execute("ROLLBACK");}catch(SQLException rollback){failure.addSuppressed(rollback);}
                throw failure;
            }
        }
    }
    private static void verifyComplete(Connection c,PrototypeCredentialCodec codec,List<Row> rows)throws SQLException,IOException{
        try(var s=c.createStatement();var r=s.executeQuery("SELECT count(*) FROM prototype_credential_formats")){
            if(!r.next()||r.getLong(1)!=rows.size())throw new IOException("CREDENTIAL_MIGRATION_INCOMPLETE");
        }
        for(Row row:rows){
            try(var p=c.prepareStatement("SELECT format FROM prototype_credential_formats WHERE kind=? AND owner_id=?")){
                p.setString(1,row.kind());p.setLong(2,row.id());try(var r=p.executeQuery()){
                    if(!r.next() || row.missing()!= "MISSING".equals(r.getString(1)))throw new IOException("CREDENTIAL_FORMAT_INVALID");
                }
            }
            PrototypeCredentialReader.read(c,codec,row.kind(),row.id(),row.value());
        }
    }
    private static boolean tableExists(Connection c)throws SQLException{
        try(var s=c.createStatement();var r=s.executeQuery("SELECT count(*) FROM sqlite_master WHERE type='table' AND name='prototype_credential_formats'")){return r.next()&&r.getInt(1)==1;}
    }
    private static boolean tableExistsNamed(Connection c,String name)throws SQLException{
        try(var p=c.prepareStatement("SELECT count(*) FROM sqlite_master WHERE type='table' AND name=?")){p.setString(1,name);try(var r=p.executeQuery()){return r.next()&&r.getInt(1)==1;}}
    }
    private static List<Row> readRows(Connection c)throws SQLException,IOException{
        List<Row> rows=new ArrayList<>();
        try(var s=c.createStatement();var r=s.executeQuery("SELECT id,api_key FROM shops ORDER BY id")){
            while(r.next())rows.add(new Row("SHOP_API_KEY",r.getLong(1),r.getLong(1),r.getString(2),r.getString(2),false));
        }
        try(var s=c.createStatement();var r=s.executeQuery("SELECT shop_id,oms_connection FROM znack_settings ORDER BY shop_id")){
            while(r.next())rows.add(new Row("ZNACK_SETTINGS_OMS",r.getLong(1),r.getLong(1),r.getString(2),r.getString(2),false));
        }
        if(tableExistsNamed(c,"znack_pipeline_connections")) try(var s=c.createStatement();var r=s.executeQuery("SELECT pipeline_id,shop_id,settings_json FROM znack_pipeline_connections ORDER BY pipeline_id")){
            while(r.next()){
                String source=r.getString(3);JsonObject json=parse(source);
                boolean missing=!json.has("omsConnection");
                var value=json.get("omsConnection");
                if(!missing&&!value.isJsonNull()&&(!value.isJsonPrimitive()||!value.getAsJsonPrimitive().isString()))throw new IOException("PIPELINE_PROFILE_INVALID");
                rows.add(new Row("PIPELINE_OMS",r.getLong(1),r.getLong(2),source,missing||value.isJsonNull()?null:value.getAsString(),missing));
            }
        }
        return rows;
    }
    private static JsonObject parse(String source)throws IOException{
        try{var parsed=JsonParser.parseString(source);if(!parsed.isJsonObject())throw new IOException("PIPELINE_PROFILE_INVALID");return parsed.getAsJsonObject();}
        catch(RuntimeException malformed){throw new IOException("PIPELINE_PROFILE_INVALID");}
    }
    private static String replaceOms(String source,String value)throws IOException{
        JsonObject object=parse(source);object.addProperty("omsConnection",value);return object.toString();
    }
}
