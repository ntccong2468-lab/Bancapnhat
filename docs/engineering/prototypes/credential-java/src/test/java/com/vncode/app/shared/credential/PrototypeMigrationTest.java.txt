package com.vncode.app.shared.credential;

import org.junit.jupiter.api.Test;
import java.io.IOException;
import java.sql.*;
import java.util.Base64;
import static org.junit.jupiter.api.Assertions.*;

class PrototypeMigrationTest {
    static final class FakeCodec implements PrototypeCredentialCodec {
        public String protect(String plain) throws IOException {
            if (plain.equals("FAIL-SYNTHETIC")) throw new IOException("synthetic failure");
            return "fake:"+Base64.getEncoder().encodeToString(plain.getBytes(java.nio.charset.StandardCharsets.UTF_8));
        }
        public String reveal(String value) throws IOException {
            if(value==null || !value.startsWith("fake:"))throw new IOException("invalid synthetic cipher");
            try{return new String(Base64.getDecoder().decode(value.substring(5)), java.nio.charset.StandardCharsets.UTF_8);}
            catch(IllegalArgumentException e){throw new IOException("invalid synthetic cipher");}
        }
    }
    private static Connection fixture() throws SQLException {
        Connection c=DriverManager.getConnection("jdbc:sqlite::memory:");
        try(Statement s=c.createStatement()){
            s.execute("CREATE TABLE shops(id INTEGER PRIMARY KEY,api_key TEXT)");
            s.execute("CREATE TABLE znack_settings(shop_id INTEGER PRIMARY KEY,oms_connection TEXT)");
            s.execute("CREATE TABLE znack_pipeline_connections(pipeline_id INTEGER PRIMARY KEY,shop_id INTEGER,settings_json TEXT)");
            s.execute("INSERT INTO shops VALUES(1,'dpapi:legacy-token'),(2,'')");
            s.execute("INSERT INTO znack_settings VALUES(1,'new-oms'),(2,NULL)");
            s.execute("INSERT INTO znack_pipeline_connections VALUES(11,1,'{\"omsConnection\":\"old-oms\",\"other\":\"retain\"}'),(12,1,'{\"omsConnection\":\"new-oms\"}')");
        }
        return c;
    }
    @Test void migratesAllAndPreservesHistoryAndTypedIdentity() throws Exception {
        try(Connection c=fixture()){
            var codec=new FakeCodec(); PrototypeCredentialMigrationService.apply(c,codec);
            try(Statement s=c.createStatement(); ResultSet r=s.executeQuery("SELECT api_key FROM shops WHERE id=1")){
                assertTrue(r.next()); assertEquals("dpapi:legacy-token",PrototypeCredentialReader.read(c,codec,"SHOP_API_KEY",1,r.getString(1)));
            }
            try(Statement s=c.createStatement(); ResultSet r=s.executeQuery("SELECT json_extract(settings_json,'$.omsConnection'),json_extract(settings_json,'$.other') FROM znack_pipeline_connections WHERE pipeline_id=11")){
                assertTrue(r.next()); assertEquals("old-oms",PrototypeCredentialReader.read(c,codec,"PIPELINE_OMS",11,r.getString(1))); assertEquals("retain",r.getString(2));
            }
            assertNull(PrototypeCredentialReader.read(c,codec,"ZNACK_SETTINGS_OMS",2,null));
            assertEquals("",PrototypeCredentialReader.read(c,codec,"SHOP_API_KEY",2,""));
            PrototypeCredentialMigrationService.apply(c,codec);
        }
    }
    @Test void refusesLegacyOrTamperedMetadata() throws Exception {
        try(Connection c=fixture()){
            var codec=new FakeCodec();
            assertThrows(SQLException.class,()->PrototypeCredentialReader.read(c,codec,"SHOP_API_KEY",1,"dpapi:legacy-token"));
            PrototypeCredentialMigrationService.apply(c,codec);
            assertThrows(IOException.class,()->PrototypeCredentialReader.read(c,codec,"SHOP_API_KEY",1,"not-a-cipher"));
            assertThrows(IOException.class,()->PrototypeCredentialReader.read(c,codec,"SHOP_API_KEY",2,"nonempty"));
        }
    }
    @Test void protectFailureDoesNotChangeAnyRowsOrMetadata() throws Exception {
        try(Connection c=fixture(); Statement s=c.createStatement()){
            s.execute("UPDATE znack_settings SET oms_connection='FAIL-SYNTHETIC' WHERE shop_id=1");
            assertThrows(IOException.class,()->PrototypeCredentialMigrationService.apply(c,new FakeCodec()));
            try(ResultSet r=s.executeQuery("SELECT api_key FROM shops WHERE id=1")){assertTrue(r.next());assertEquals("dpapi:legacy-token",r.getString(1));}
            try(ResultSet r=s.executeQuery("SELECT COUNT(*) FROM sqlite_master WHERE name='prototype_credential_formats'")){assertTrue(r.next());assertEquals(0,r.getInt(1));}
        }
    }
    @Test void changedSourceAfterProtectionIsNotOverwritten() throws Exception {
        try(Connection c=fixture()){
            FakeCodec delegate=new FakeCodec();
            PrototypeCredentialCodec racing=new PrototypeCredentialCodec(){
                boolean changed;
                public String protect(String plain)throws IOException {
                    if(!changed){try(Statement s=c.createStatement()){s.execute("UPDATE znack_settings SET oms_connection='rotated' WHERE shop_id=1");changed=true;}catch(SQLException e){throw new IOException("synthetic failure");}}
                    return delegate.protect(plain);
                }
                public String reveal(String value)throws IOException{return delegate.reveal(value);}
            };
            assertThrows(SQLException.class,()->PrototypeCredentialMigrationService.apply(c,racing));
            try(Statement s=c.createStatement();ResultSet r=s.executeQuery("SELECT oms_connection FROM znack_settings WHERE shop_id=1")){assertTrue(r.next());assertEquals("rotated",r.getString(1));}
        }
    }
    @Test void transactionInterruptionRollsBackAllValuesAndMetadata() throws Exception {
        try(Connection c=fixture();Statement s=c.createStatement()){
            s.execute("CREATE TRIGGER simulate_interruption BEFORE UPDATE ON znack_pipeline_connections BEGIN SELECT RAISE(ABORT,'synthetic interruption'); END");
            assertThrows(SQLException.class,()->PrototypeCredentialMigrationService.apply(c,new FakeCodec()));
            try(ResultSet r=s.executeQuery("SELECT api_key FROM shops WHERE id=1")){assertTrue(r.next());assertEquals("dpapi:legacy-token",r.getString(1));}
            try(ResultSet r=s.executeQuery("SELECT COUNT(*) FROM sqlite_master WHERE name='prototype_credential_formats'")){assertTrue(r.next());assertEquals(0,r.getInt(1));}
        }
    }
    @Test void systemCodecFailsClosedOutsideWindows() throws Exception {
        org.junit.jupiter.api.Assumptions.assumeFalse(com.vncode.app.shared.WindowsSecretProtector.supported());
        try(Connection c=fixture()){
            assertThrows(IOException.class,()->PrototypeCredentialMigrationService.apply(c,PrototypeCredentialAccess.codec()));
        }
    }
    @Test void allowsLazyPipelineTableToBeAbsent() throws Exception {
        try(Connection c=fixture();Statement s=c.createStatement()){
            s.execute("DROP TABLE znack_pipeline_connections");
            PrototypeCredentialMigrationService.apply(c,new FakeCodec());
            PrototypeCredentialMigrationService.apply(c,new FakeCodec());
        }
    }
    @Test void missingOmsAttributePreservesJsonAndHasExplicitMetadata() throws Exception {
        try(Connection c=fixture();Statement s=c.createStatement()){
            s.execute("UPDATE znack_pipeline_connections SET settings_json='{}' WHERE pipeline_id=11");
            PrototypeCredentialMigrationService.apply(c,new FakeCodec());
            try(ResultSet r=s.executeQuery("SELECT settings_json FROM znack_pipeline_connections WHERE pipeline_id=11")){assertTrue(r.next());assertEquals("{}",r.getString(1));}
            assertNull(PrototypeCredentialReader.read(c,new FakeCodec(),"PIPELINE_OMS",11,null));
            PrototypeCredentialMigrationService.apply(c,new FakeCodec());
        }
    }
    @Test void rejectsPartialMigrationAndInvalidJson() throws Exception {
        try(Connection c=fixture(); Statement s=c.createStatement()){
            s.execute("UPDATE znack_pipeline_connections SET settings_json='bad json' WHERE pipeline_id=11");
            assertThrows(IOException.class,()->PrototypeCredentialMigrationService.apply(c,new FakeCodec()));
            try(ResultSet r=s.executeQuery("SELECT api_key FROM shops WHERE id=1")){assertTrue(r.next());assertEquals("dpapi:legacy-token",r.getString(1));}
        }
    }
}
