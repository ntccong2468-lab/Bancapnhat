package com.vncode.app.shared.credential;

import com.vncode.app.config.Database;
import com.vncode.app.features.kiz.KizService;
import com.vncode.app.features.license.LicenseDomainFixture;
import com.vncode.app.features.license.LicenseRuntime;
import com.vncode.app.integration.listing.ListingIdentity;
import com.vncode.app.integration.marketplace.Marketplace;
import com.vncode.app.models.Shop;
import java.io.IOException;
import java.nio.file.Path;
import java.sql.Connection;
import java.sql.SQLException;
import java.util.HashMap;
import java.util.Map;
import org.junit.jupiter.api.AfterEach;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.io.TempDir;
import static org.junit.jupiter.api.Assertions.*;

/** Prototype tests only: fake values and a reversible in-memory codec, never production DPAPI. */
class PrototypeCredentialConsumerContractTest {
    @TempDir Path directory;
    private String prior;
    private AutoCloseable codecScope;
    private final FakeCodec codec = new FakeCodec();

    @BeforeEach void setup() throws Exception {
        prior = System.getProperty("vncode.appdata.dir");
        System.setProperty("vncode.appdata.dir", directory.toString());
        Database.initDatabase();
        codecScope = PrototypeCredentialAccess.withCodecForTest(codec);
        try (var c = Database.getConnection(); var s = c.createStatement()) {
            s.execute("CREATE TABLE prototype_credential_formats(kind TEXT NOT NULL, owner_id INTEGER NOT NULL, format TEXT NOT NULL, PRIMARY KEY(kind,owner_id))");
        }
    }

    @AfterEach void cleanup() throws Exception {
        codecScope.close();
        if (prior == null) System.clearProperty("vncode.appdata.dir");
        else System.setProperty("vncode.appdata.dir", prior);
    }

    @Test void decodedSqlConsumerPreservesOwnerAndDetectsRotation() throws Exception {
        String token = "  dpapi:synthetic  ";
        var expected = new Shop(7, "Fixture", Marketplace.OZON, "client-7", token);
        String stored = codec.protect(token);
        insert(7, "OZON", "client-7", stored, "PROTECTED");
        try (var c = Database.getConnection()) {
            String decoded = PrototypeCredentialReader.read(c, codec, "SHOP_API_KEY", 7, stored);
            assertEquals(token, decoded);
            assertEquals(ListingIdentity.fingerprint(expected), ListingIdentity.fingerprint(
                    new Shop(7, "Fixture", Marketplace.OZON, "client-7", decoded)));
            assertEquals(ListingIdentity.fingerprint(expected),
                    ListingIdentity.fingerprint(c, 7, Marketplace.OZON),
                    "SQL owner proof must be based on the decrypted logical credential");
        }
        assertDoesNotThrow(() -> ListingIdentity.require(expected));
        try (var c = Database.getConnection(); var p = c.prepareStatement("UPDATE shops SET api_key=? WHERE id=7")) {
            p.setString(1, codec.protect("new-synthetic-token")); p.executeUpdate();
        }
        assertThrows(SecurityException.class, () -> ListingIdentity.require(expected));
    }

    @Test void differentCiphertextsForSameWbTokenRemainAmbiguousAfterDecoding() throws Exception {
        String token = "synthetic-duplicate";
        String first = codec.protect(token), second = codec.protect(token);
        assertNotEquals(first, second);
        insert(1, "WILDBERRIES", null, first, "PROTECTED");
        insert(2, "WILDBERRIES", null, second, "PROTECTED");
        try (var c = Database.getConnection()) {
            assertEquals(2, countDecodedWbMatches(c, token));
        }
        assertDoesNotThrow(() -> KizService.requireCurrentShop(1, token));
        var previousLicense = LicenseRuntime.service();
        try (var license = LicenseDomainFixture.active(directory.resolve("license-fixture"))) {
            LicenseRuntime.install(license);
            assertEquals("SHOP_CREDENTIALS_AMBIGUOUS", assertThrows(IOException.class,
                    () -> KizService.addDataMatrixCodeToOrder(token, 1L, "synthetic-code")).getMessage());
        } finally {
            LicenseRuntime.install(previousLicense);
        }
    }

    @Test void missingOrUnreadableCredentialsCannotBeTreatedAsNonmatchingShops() throws Exception {
        String stored = codec.protect("synthetic-token");
        insert(1, "WILDBERRIES", null, stored, "PROTECTED");
        insert(2, "WILDBERRIES", null, "synthetic-legacy", null);
        try (var c = Database.getConnection()) {
            var failure = assertThrows(IOException.class, () -> countDecodedWbMatches(c, "synthetic-token"));
            assertEquals("CREDENTIAL_FORMAT_UNAVAILABLE", failure.getMessage());
        }
        try (var c = Database.getConnection(); var p = c.prepareStatement(
                "INSERT INTO prototype_credential_formats(kind,owner_id,format) VALUES('SHOP_API_KEY',2,'PROTECTED')")) {
            p.executeUpdate();
        }
        try (var c = Database.getConnection()) {
            var failure = assertThrows(IOException.class, () -> countDecodedWbMatches(c, "synthetic-token"));
            assertTrue("CREDENTIAL_PROTECTION_FAILED".equals(failure.getMessage())
                    || "SYNTHETIC_CODEC_UNAVAILABLE".equals(failure.getMessage()));
        }
        assertEquals("SHOP_CREDENTIALS_UNAVAILABLE", assertThrows(IOException.class,
                () -> KizService.requireCurrentShop(2, "synthetic-token")).getMessage());
    }

    private int countDecodedWbMatches(Connection c, String token) throws SQLException, IOException {
        int count = 0;
        try (var p = c.prepareStatement("SELECT id,api_key FROM shops WHERE marketplace='WILDBERRIES'"); var rows = p.executeQuery()) {
            while (rows.next()) {
                String decoded = PrototypeCredentialReader.read(c, codec, "SHOP_API_KEY", rows.getLong(1), rows.getString(2));
                if (token.equals(decoded)) count++;
            }
        }
        return count;
    }

    private void insert(int id, String marketplace, String client, String stored, String format) throws SQLException {
        try (var c = Database.getConnection()) {
            try (var p = c.prepareStatement("INSERT INTO shops(id,name,marketplace,client_id,api_key) VALUES(?,?,?,?,?)")) {
                p.setInt(1, id); p.setString(2, "Fixture"); p.setString(3, marketplace); p.setString(4, client); p.setString(5, stored); p.executeUpdate();
            }
            if (format != null) try (var p = c.prepareStatement(
                    "INSERT INTO prototype_credential_formats(kind,owner_id,format) VALUES('SHOP_API_KEY',?,?)")) {
                p.setInt(1, id); p.setString(2, format); p.executeUpdate();
            }
        }
    }

    private static final class FakeCodec implements PrototypeCredentialCodec {
        private final Map<String, String> values = new HashMap<>();
        public String protect(String plain) {
            String marker = "synthetic-protected-" + (values.size() + 1);
            values.put(marker, plain);
            return marker;
        }
        public String reveal(String protectedValue) throws IOException {
            String plain = values.get(protectedValue);
            if (plain == null) throw new IOException("SYNTHETIC_CODEC_UNAVAILABLE");
            return plain;
        }
    }
}
