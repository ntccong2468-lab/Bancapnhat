package com.vncode.app.shared.credential;

import com.vncode.app.shared.WindowsSecretProtector;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.condition.EnabledOnOs;
import org.junit.jupiter.api.condition.OS;
import java.io.IOException;
import static org.junit.jupiter.api.Assertions.*;

/** Synthetic only: requires actual Windows and a non-production CurrentUser profile. */
@EnabledOnOs(OS.WINDOWS)
class PrototypeWindowsDpapiTest {
    @Test void syntheticCurrentUserRoundTripAndPurposeSeparation() throws Exception {
        var prototype=PrototypeCredentialAccess.codec();
        prototype.ensureAvailable();
        String testSecret="synthetic-only-Юникод-"+"x".repeat(16384);
        String ciphertext=prototype.protect(testSecret);
        assertEquals(testSecret,prototype.reveal(ciphertext));
        var otherPurpose=new WindowsSecretProtector(WindowsSecretProtector.CATALOG_PURPOSE,8192);
        assertThrows(IOException.class,()->otherPurpose.reveal(ciphertext));
        assertThrows(IOException.class,()->prototype.reveal("dpapi:AQIDBA=="));
    }
}
