package com.vncode.app.shared;

import static org.junit.jupiter.api.Assertions.*;

import com.vncode.app.shared.credential.PrototypeCredentialCodec;
import com.vncode.app.shared.credential.PrototypeCredentialMigrationService;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;
import java.util.Base64;
import java.util.concurrent.atomic.AtomicBoolean;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.io.TempDir;

/** Test-only gate: it is deliberately not wired into application startup. */
class CredentialMigrationGateFixtureTest {
    @TempDir Path temporaryDirectory;

    @Test void verifiedSnapshotAllowsMigrationAndExplicitRestoreRecoversLegacy() throws Exception {
        Path root = Files.createDirectory(temporaryDirectory.resolve("success"));
        Path database = root.resolve("database.db");
        createFixture(database);
        LocalDataSnapshotService.Snapshot snapshot = runGate(root, s -> {}, new FixtureCodec());
        assertEquals("PROTECTED", query(database, "SELECT format FROM prototype_credential_formats WHERE kind='SHOP_API_KEY' AND owner_id=1"));
        String migrated = query(database, "SELECT api_key FROM shops WHERE id=1");
        assertNotEquals("fixture-shop", migrated);
        runGate(root, s -> {}, new FixtureCodec() {
            @Override public String protect(String plain) throws IOException {
                throw new IOException("Already migrated data must not be encrypted again");
            }
        });
        assertEquals(migrated, query(database, "SELECT api_key FROM shops WHERE id=1"));
        try (var owner = AppDataLock.acquire(root, "fixture")) {
            var service = new LocalDataSnapshotService();
            assertTrue(service.verify(snapshot));
            var restored = service.restore(owner, snapshot, "1.6.0");
            assertTrue(service.verify(restored.recoveryBundle()));
            assertEquals("PROTECTED", query(restored.recoveryBundle().database(), "SELECT format FROM prototype_credential_formats WHERE kind='SHOP_API_KEY' AND owner_id=1"));
        }
        assertLegacy(database);
    }

    @Test void failedEncryptionDoesNotCommitAndRetainsVerifiedSnapshot() throws Exception {
        Path root = Files.createDirectory(temporaryDirectory.resolve("failed-codec"));
        Path database = root.resolve("database.db");
        createFixture(database);
        var codec = new FixtureCodec() {
            private int calls;
            @Override public String protect(String plain) throws IOException {
                if (++calls == 2) throw new IOException("synthetic codec failure");
                return super.protect(plain);
            }
        };
        assertThrows(IOException.class, () -> runGate(root, s -> {}, codec));
        assertLegacy(database);
        try (var owner = AppDataLock.acquire(root, "fixture")) {
            var service = new LocalDataSnapshotService();
            var snapshots = service.list(owner);
            assertEquals(1, snapshots.size());
            assertTrue(service.verify(snapshots.getFirst()));
        }
    }

    @Test void snapshotCorruptionPreventsAnyCredentialTransformation() throws Exception {
        Path root = Files.createDirectory(temporaryDirectory.resolve("invalid-snapshot"));
        Path database = root.resolve("database.db");
        createFixture(database);
        var used = new AtomicBoolean();
        var codec = new FixtureCodec() {
            @Override public String protect(String plain) throws IOException {
                used.set(true);
                return super.protect(plain);
            }
        };
        assertThrows(IOException.class, () -> runGate(root,
                snapshot -> Files.write(snapshot.database(), new byte[] {1}, StandardOpenOption.APPEND), codec));
        assertFalse(used.get());
        assertLegacy(database);
    }

    private static LocalDataSnapshotService.Snapshot runGate(Path root, SnapshotHook hook, PrototypeCredentialCodec codec) throws Exception {
        try (var owner = AppDataLock.acquire(root, "fixture")) {
            var service = new LocalDataSnapshotService();
            service.recoverInterrupted(owner);
            var snapshot = service.create(owner, "1.6.0", 6, "synthetic-credential-migration");
            hook.afterCreate(snapshot);
            if (!service.verify(snapshot)) throw new IOException("Snapshot verification failed");
            try (var connection = connect(root.resolve("database.db"))) {
                PrototypeCredentialMigrationService.apply(connection, codec);
            }
            return snapshot;
        }
    }

    @FunctionalInterface private interface SnapshotHook {
        void afterCreate(LocalDataSnapshotService.Snapshot snapshot) throws IOException;
    }

    private static class FixtureCodec implements PrototypeCredentialCodec {
        @Override public String protect(String plain) throws IOException {
            return "fixture:" + Base64.getEncoder().encodeToString(plain.getBytes(StandardCharsets.UTF_8));
        }
        @Override public String reveal(String value) throws IOException {
            if (!value.startsWith("fixture:")) throw new IOException("bad fixture");
            return new String(Base64.getDecoder().decode(value.substring(8)), StandardCharsets.UTF_8);
        }
    }

    private static void createFixture(Path database) throws SQLException {
        try (var connection = connect(database); var sql = connection.createStatement()) {
            sql.execute("CREATE TABLE shops(id INTEGER PRIMARY KEY,api_key TEXT NOT NULL)");
            sql.execute("CREATE TABLE znack_settings(shop_id INTEGER PRIMARY KEY,oms_connection TEXT)");
            sql.execute("CREATE TABLE znack_pipeline_connections(pipeline_id INTEGER PRIMARY KEY,shop_id INTEGER,settings_json TEXT)");
            sql.execute("INSERT INTO shops VALUES(1,'fixture-shop')");
            sql.execute("INSERT INTO znack_settings VALUES(1,'fixture-current')");
            sql.execute("INSERT INTO znack_pipeline_connections VALUES(101,1,json_object('omsConnection','fixture-old')),(102,1,json_object('omsConnection','fixture-new'))");
        }
    }

    private static void assertLegacy(Path database) throws SQLException {
        assertEquals("fixture-shop", query(database, "SELECT api_key FROM shops WHERE id=1"));
        assertEquals("fixture-current", query(database, "SELECT oms_connection FROM znack_settings WHERE shop_id=1"));
        assertEquals("fixture-old", query(database, "SELECT json_extract(settings_json,'$.omsConnection') FROM znack_pipeline_connections WHERE pipeline_id=101"));
        assertEquals("fixture-new", query(database, "SELECT json_extract(settings_json,'$.omsConnection') FROM znack_pipeline_connections WHERE pipeline_id=102"));
        assertEquals("0", query(database, "SELECT count(*) FROM sqlite_master WHERE type='table' AND name='prototype_credential_formats'"));
    }

    private static Connection connect(Path database) throws SQLException {
        return DriverManager.getConnection("jdbc:sqlite:" + database.toAbsolutePath());
    }

    private static String query(Path database, String sql) throws SQLException {
        try (var connection = connect(database); var statement = connection.createStatement(); var result = statement.executeQuery(sql)) {
            assertTrue(result.next());
            return result.getString(1);
        }
    }
}
